Register a new module

Add a module by creating a loader in system_manager/ that imports the target Flask app and returns a Blueprint.

# system_manager/new_module.py
from flask import Blueprint

def new_module_blueprint():
    bp = Blueprint("new_module", __name__, url_prefix="/new")
    @bp.route("/")
    def index():
        return "Hello from new module"
    return bp

# Then in system_manager/__init__.py:
